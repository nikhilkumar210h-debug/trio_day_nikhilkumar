rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function owner(uid) { return signedIn() && request.auth.uid == uid; }
    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/config/admins)
        && request.auth.uid in get(/databases/$(database)/documents/config/admins).data.uids;
    }

    // Stories and voice statuses share the historical `posts` collection name.
    // The retired Post/Feed type is no longer readable or creatable.
    match /posts/{postId} {
      allow read: if signedIn() && (
        (
          resource.data.type == 'story'
          && resource.data.isStory == true
          && (
            resource.data.privacy == 'public'
            || resource.data.uid == request.auth.uid
            || (
              resource.data.privacy == 'friends'
              && resource.data.allowedUids is list
              && request.auth.uid in resource.data.allowedUids
            )
          )
        )
        || (
          resource.data.type == 'voice'
          && resource.data.isVoice == true
          && (
            resource.data.privacy == 'public'
            || resource.data.uid == request.auth.uid
            || (
              resource.data.privacy == 'friends'
              && resource.data.allowedUids is list
              && request.auth.uid in resource.data.allowedUids
            )
          )
        )
      );

      function canReadStory(storyId) {
        let s = get(/databases/$(database)/documents/posts/$(storyId)).data;
        return (
          (s.type == 'story' && s.isStory == true)
          || (s.type == 'voice' && s.isVoice == true)
        ) && (
          s.privacy == 'public'
          || s.uid == request.auth.uid
          || (
            s.privacy == 'friends'
            && s.allowedUids is list
            && request.auth.uid in s.allowedUids
          )
        );
      }

      function validStoryCreate() {
        return request.resource.data.isStory == true
          && request.resource.data.privacy in ['public', 'friends']
          && request.resource.data.expiresAtMs is number
          && request.resource.data.expiresAtMs >= request.time.toMillis() + 23 * 60 * 60 * 1000
          && request.resource.data.expiresAtMs <= request.time.toMillis() + 25 * 60 * 60 * 1000
          && (!('mediaUrl' in request.resource.data) || request.resource.data.mediaUrl is string)
          && (!('editorMeta' in request.resource.data) || (request.resource.data.editorMeta is map && request.resource.data.editorMeta.size() <= 6))
          && (
            request.resource.data.privacy == 'public'
            || (
              request.resource.data.allowedUids is list
              && request.resource.data.allowedUids.size() > 0
              && request.resource.data.allowedUids.size() <= 500
              && request.auth.uid in request.resource.data.allowedUids
            )
          );
      }

      function validVoiceCreate() {
        return request.resource.data.isVoice == true
          && request.resource.data.duration is number
          && request.resource.data.duration > 0
          && request.resource.data.duration <= 60
          && request.resource.data.privacy in ['public', 'friends']
          && request.resource.data.expiresAtMs is number
          && request.resource.data.mediaUrl is string
          && (
            request.resource.data.privacy == 'public'
            || (
              request.resource.data.allowedUids is list
              && request.resource.data.allowedUids.size() <= 500
              && request.auth.uid in request.resource.data.allowedUids
            )
          );
      }

      allow create: if signedIn()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 50
        && request.resource.data.message is string
        && request.resource.data.message.size() > 0
        && request.resource.data.message.size() <= 500
        && request.resource.data.type in ['story', 'voice']
        && (
          (request.resource.data.type == 'story' && validStoryCreate())
          || (request.resource.data.type == 'voice' && validVoiceCreate())
        );

      allow update: if false;
      allow delete: if signedIn() && resource.data.uid == request.auth.uid;

      match /comments/{commentId} {
        allow read: if signedIn() && canReadStory(postId);
        allow create: if signedIn()
          && canReadStory(postId)
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.txt is string
          && request.resource.data.txt.size() > 0
          && request.resource.data.txt.size() < 200
          && request.resource.data.name is string;
        allow update, delete: if false;
      }

      match /likes/{likeUid} {
        allow read: if signedIn() && canReadStory(postId);
        allow create: if signedIn()
          && request.auth.uid == likeUid
          && canReadStory(postId)
          && request.resource.data.uid == request.auth.uid;
        allow delete: if signedIn() && request.auth.uid == likeUid;
        allow update: if false;
      }

      match /moods/{moodUid} {
        allow read: if signedIn() && canReadStory(postId);
        allow create: if signedIn()
          && request.auth.uid == moodUid
          && canReadStory(postId)
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.mood is string;
        allow delete: if signedIn() && request.auth.uid == moodUid;
        allow update: if signedIn() && request.auth.uid == moodUid;
      }
    }

    // Gamification fields that ONLY Cloud Functions / server logic may write.
    // A client-side update that touches any of these keys is rejected.
    function noGamificationWrite() {
      return !request.resource.data.diff(resource.data).affectedKeys()
        .hasAny([
          'xp', 'level', 'streakCurrent', 'streakBest', 'streakLastDate',
          'weeklyXp', 'monthlyXp', 'badges'
        ]);
    }

    // Allow userId change only if current value is missing (first-time set).
    function canChangeUserId() {
      return !('userId' in resource.data)
        || resource.data.userId == null
        || resource.data.userId == '';
    }

    function validFourOptions(options) {
      return options is list
        && options.size() == 4
        && options[0] is string && options[0].size() > 0 && options[0].size() <= 180
        && options[1] is string && options[1].size() > 0 && options[1].size() <= 180
        && options[2] is string && options[2].size() > 0 && options[2].size() <= 180
        && options[3] is string && options[3].size() > 0 && options[3].size() <= 180
        && options[0] != options[1]
        && options[0] != options[2]
        && options[0] != options[3]
        && options[1] != options[2]
        && options[1] != options[3]
        && options[2] != options[3];
    }

    function validQuizInteraction(activityType, value) {
      return value is map
        && value.kind == 'quiz'
        && value.question is string
        && value.question.size() >= 10
        && value.question.size() <= 500
        && validFourOptions(value.options)
        && value.correct is int
        && value.correct >= 0
        && value.correct <= 3
        && value.proofRequired == true
        && value.proofPrompt is string
        && value.proofPrompt.size() >= 10
        && value.proofPrompt.size() <= 240
        && (
          activityType == 'puzzle'
          || (
            activityType == 'learn'
            && value.lesson is string
            && value.lesson.size() >= 20
            && value.lesson.size() <= 900
          )
        );
    }

    function validAllocationItems(items) {
      return items is list
        && items.size() >= 2
        && items.size() <= 8
        && items[0] is list && items[0].size() == 2
        && items[0][0] is string && items[0][0].size() > 0 && items[0][0].size() <= 50
        && items[0][1] is number && items[0][1] >= 0
        && items[1] is list && items[1].size() == 2
        && items[1][0] is string && items[1][0].size() > 0 && items[1][0].size() <= 50
        && items[1][1] is number && items[1][1] >= 0
        && (items.size() < 3 || (
          items[2] is list && items[2].size() == 2
          && items[2][0] is string && items[2][0].size() > 0 && items[2][0].size() <= 50
          && items[2][1] is number && items[2][1] >= 0
        ))
        && (items.size() < 4 || (
          items[3] is list && items[3].size() == 2
          && items[3][0] is string && items[3][0].size() > 0 && items[3][0].size() <= 50
          && items[3][1] is number && items[3][1] >= 0
        ))
        && (items.size() < 5 || (
          items[4] is list && items[4].size() == 2
          && items[4][0] is string && items[4][0].size() > 0 && items[4][0].size() <= 50
          && items[4][1] is number && items[4][1] >= 0
        ))
        && (items.size() < 6 || (
          items[5] is list && items[5].size() == 2
          && items[5][0] is string && items[5][0].size() > 0 && items[5][0].size() <= 50
          && items[5][1] is number && items[5][1] >= 0
        ))
        && (items.size() < 7 || (
          items[6] is list && items[6].size() == 2
          && items[6][0] is string && items[6][0].size() > 0 && items[6][0].size() <= 50
          && items[6][1] is number && items[6][1] >= 0
        ))
        && (items.size() < 8 || (
          items[7] is list && items[7].size() == 2
          && items[7][0] is string && items[7][0].size() > 0 && items[7][0].size() <= 50
          && items[7][1] is number && items[7][1] >= 0
        ));
    }

    function validBuildInteraction(value) {
      return value is map
        && value.kind == 'build'
        && value.mechanic in ['order','allocate','grid','assign']
        && (
          (
            value.mechanic == 'order'
            && value.items is list
            && value.items.size() >= 3
            && value.items.size() <= 8
            && value.items[0] is string && value.items[0].size() > 0 && value.items[0].size() <= 70
            && value.items[1] is string && value.items[1].size() > 0 && value.items[1].size() <= 70
          )
          || (
            value.mechanic == 'allocate'
            && value.budget is number
            && value.budget >= 1
            && value.budget <= 100000
            && validAllocationItems(value.items)
          )
          || (
            value.mechanic == 'grid'
            && value.size == 4
            && value.required is list
            && value.required.size() >= 2
            && value.required.size() <= 4
            && value.required[0] is string && value.required[0].size() > 0 && value.required[0].size() <= 40
            && value.required[1] is string && value.required[1].size() > 0 && value.required[1].size() <= 40
            && (value.required.size() < 3 || (value.required[2] is string && value.required[2].size() > 0 && value.required[2].size() <= 40))
            && (value.required.size() < 4 || (value.required[3] is string && value.required[3].size() > 0 && value.required[3].size() <= 40))
            && value.blocked is list
            && value.blocked.size() <= 8
            && value.adjacentPairs is list
            && value.adjacentPairs.size() <= 3
          )
          || (
            value.mechanic == 'assign'
            && value.people is list
            && value.people.size() == 4
            && value.roles is list
            && value.roles.size() == 4
            && value.people[0] is string && value.people[0].size() > 0 && value.people[0].size() <= 40
            && value.people[1] is string && value.people[1].size() > 0 && value.people[1].size() <= 40
            && value.people[2] is string && value.people[2].size() > 0 && value.people[2].size() <= 40
            && value.people[3] is string && value.people[3].size() > 0 && value.people[3].size() <= 40
            && value.roles[0] is string && value.roles[0].size() > 0 && value.roles[0].size() <= 40
            && value.roles[1] is string && value.roles[1].size() > 0 && value.roles[1].size() <= 40
            && value.roles[2] is string && value.roles[2].size() > 0 && value.roles[2].size() <= 40
            && value.roles[3] is string && value.roles[3].size() > 0 && value.roles[3].size() <= 40
            && value.correct is map
            && value.correct[value.people[0]] is string && value.correct[value.people[0]] in value.roles
            && value.correct[value.people[1]] is string && value.correct[value.people[1]] in value.roles
            && value.correct[value.people[2]] is string && value.correct[value.people[2]] in value.roles
            && value.correct[value.people[3]] is string && value.correct[value.people[3]] in value.roles
            && value.correct[value.people[0]] != value.correct[value.people[1]]
            && value.correct[value.people[0]] != value.correct[value.people[2]]
            && value.correct[value.people[0]] != value.correct[value.people[3]]
            && value.correct[value.people[1]] != value.correct[value.people[2]]
            && value.correct[value.people[1]] != value.correct[value.people[3]]
            && value.correct[value.people[2]] != value.correct[value.people[3]]
          )
        );
    }

    function validChallengeRound(round) {
      return round is map
        && round.q is string
        && round.q.size() >= 8
        && round.q.size() <= 240
        && validFourOptions(round.o)
        && round.a is int
        && round.a >= 0
        && round.a <= 3;
    }

    function validChallengeInteraction(value) {
      return value is map
        && value.kind == 'challenge'
        && value.rounds is list
        && value.rounds.size() == 5
        && validChallengeRound(value.rounds[0])
        && validChallengeRound(value.rounds[1])
        && validChallengeRound(value.rounds[2])
        && validChallengeRound(value.rounds[3])
        && validChallengeRound(value.rounds[4]);
    }

    function gridAdjacent(a, b) {
      return (a + 1 == b && a != 3 && a != 7 && a != 11)
        || (b + 1 == a && b != 3 && b != 7 && b != 11)
        || a + 4 == b
        || b + 4 == a;
    }

    function validAllocationSolution(taskId, state) {
      let d = get(/databases/$(database)/documents/communityTasks/$(taskId)).data;
      let items = d.interaction.items;
      let a = state.allocation;
      return a is list
        && a.size() == items.size()
        && a[0] is number && a[0] >= items[0][1]
        && a[1] is number && a[1] >= items[1][1]
        && (items.size() < 3 || (a[2] is number && a[2] >= items[2][1]))
        && (items.size() < 4 || (a[3] is number && a[3] >= items[3][1]))
        && (items.size() < 5 || (a[4] is number && a[4] >= items[4][1]))
        && (items.size() < 6 || (a[5] is number && a[5] >= items[5][1]))
        && (items.size() < 7 || (a[6] is number && a[6] >= items[6][1]))
        && (items.size() < 8 || (a[7] is number && a[7] >= items[7][1]))
        && a[0] + a[1]
          + (items.size() >= 3 ? a[2] : 0)
          + (items.size() >= 4 ? a[3] : 0)
          + (items.size() >= 5 ? a[4] : 0)
          + (items.size() >= 6 ? a[5] : 0)
          + (items.size() >= 7 ? a[6] : 0)
          + (items.size() >= 8 ? a[7] : 0)
          <= d.interaction.budget;
    }

    function validGridSolution(taskId, state) {
      let d = get(/databases/$(database)/documents/communityTasks/$(taskId)).data;
      let r = d.interaction.required;
      let p = state.positions;
      return p is map
        && p[r[0]] is int && p[r[0]] >= 0 && p[r[0]] < 16
        && p[r[1]] is int && p[r[1]] >= 0 && p[r[1]] < 16
        && p[r[0]] != p[r[1]]
        && !(p[r[0]] in d.interaction.blocked)
        && !(p[r[1]] in d.interaction.blocked)
        && gridAdjacent(p[r[0]], p[r[1]])
        && (r.size() < 3 || (
          p[r[2]] is int && p[r[2]] >= 0 && p[r[2]] < 16
          && p[r[2]] != p[r[0]] && p[r[2]] != p[r[1]]
          && !(p[r[2]] in d.interaction.blocked)
          && gridAdjacent(p[r[1]], p[r[2]])
        ))
        && (r.size() < 4 || (
          p[r[3]] is int && p[r[3]] >= 0 && p[r[3]] < 16
          && p[r[3]] != p[r[0]] && p[r[3]] != p[r[1]] && p[r[3]] != p[r[2]]
          && !(p[r[3]] in d.interaction.blocked)
          && gridAdjacent(p[r[2]], p[r[3]])
        ));
    }

    match /usersPrivate/{userId} {
      allow read, create, update: if owner(userId);
      allow delete: if owner(userId);
    }

    match /users/{userId} {
      allow read: if signedIn();
      allow create: if owner(userId);
      allow update: if owner(userId)
      && noGamificationWrite()
      && (!request.resource.data.diff(resource.data).affectedKeys().hasAll(['userId']) || canChangeUserId());
      allow delete: if false;
      match /followers/{followerId} {
        allow read: if signedIn();
        allow create: if owner(followerId);
        allow delete: if owner(followerId);
        allow update: if false;
      }
      match /following/{followedId} {
        allow read: if signedIn();
        allow create: if owner(userId) && followedId != request.auth.uid;
        allow delete: if owner(userId);
        allow update: if false;
      }
      match /notifications/{notificationId} {
        allow read, update: if owner(userId);
        // Notification creation is server-authoritative. Client code must use the
        // authenticated notification Worker, which verifies the triggering action.
        allow create: if false;
        allow delete: if owner(userId);
      }
      // Progress-only storage (one doc per period, not per task)
      match /progress/{periodKey} {
        allow read, write: if owner(userId);
      }
      match /activityCompletions/{completionId} {
        // Built-in completions are server-authoritative. Clients may only read
        // their own history; the gamification Worker creates verified records.
        allow read: if owner(userId);
        allow create, update, delete: if false;
      }
    }

    match /groupChat/{messageId} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid
        && request.resource.data.text is string && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 500;
      allow update: if false;
      allow delete: if signedIn() && resource.data.uid == request.auth.uid;
    }

    match /privateChats/{chatId}/messages/{messageId} {
      // chatId format: uid1_uid2 where uid1 < uid2 lexicographically
      // Allow access only if current user is one of the two participants
      allow read: if signedIn() && (chatId == request.auth.uid + '_' + request.auth.uid
        || (chatId.matches('^' + request.auth.uid + '_[^_]+$')
          || chatId.matches('^[^_]+_' + request.auth.uid + '$')));
      allow create: if signedIn() && (chatId.matches('^' + request.auth.uid + '_[^_]+$')
          || chatId.matches('^[^_]+_' + request.auth.uid + '$'))
        && request.resource.data.uid == request.auth.uid && request.resource.data.text is string
        && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 500;
      allow update: if signedIn()
        && (chatId.matches('^' + request.auth.uid + '_[^_]+$')
          || chatId.matches('^[^_]+_' + request.auth.uid + '$'))
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['seen'])
        && request.resource.data.seen is number;
      allow delete: if signedIn()
        && (chatId.matches('^' + request.auth.uid + '_[^_]+$')
          || chatId.matches('^[^_]+_' + request.auth.uid + '$'))
        && resource.data.uid == request.auth.uid;
    }

    // ── Gamification ──────────────────────────────────────────────
    match /config/{docId} {
      allow read: if signedIn();
      allow write: if false; // Console-only (admins list)
    }

    match /badges/{badgeId} {
      allow read: if signedIn();
      allow create, update: if isAdmin();
      allow delete: if isAdmin();
    }

    match /leaderboards/{boardId} {
      allow read: if signedIn();
      // Leaderboard documents are server-managed only.
      // No authenticated client may create, update, or delete them directly.
      // All leaderboard mutations must go through Cloud Functions / Admin SDK.
      allow write: if false;
    }

    match /communityTasks/{taskId} {
      // Community-created Challenges are the only custom challenge records now.
      allow read: if signedIn();

      function validChallengeInteraction() {
        return request.resource.data.interaction is map
          && request.resource.data.interaction.kind == 'choice'
          && request.resource.data.interaction.question is string
          && request.resource.data.interaction.question.size() > 0
          && request.resource.data.interaction.question.size() <= 240
          && request.resource.data.interaction.options is list
          && request.resource.data.interaction.options.size() >= 2
          && request.resource.data.interaction.options.size() <= 5;
      }

      allow create: if signedIn()
        && request.resource.data.creatorUid == request.auth.uid
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.title.size() <= 100
        && request.resource.data.kind == 'challenge'
        && request.resource.data.activityType == 'challenge'
        && request.resource.data.category is string
        && request.resource.data.category.size() > 0
        && request.resource.data.category.size() <= 60
        && request.resource.data.durationMin is number
        && request.resource.data.durationMin >= 1
        && request.resource.data.durationMin <= 30
        && request.resource.data.difficulty == 'Easy'
        && request.resource.data.endAtMs is number
        && request.resource.data.expiresInDays in [7,14,21,30,40]
        && request.resource.data.startAtMs is number
        && request.resource.data.startAtMs <= request.time.toMillis()
        && request.resource.data.endAtMs == request.resource.data.startAtMs + request.resource.data.expiresInDays * 86400000
        && request.resource.data.endAtMs > request.time.toMillis()
        && request.resource.data.status == 'active'
        && request.resource.data.featured == false
        && request.resource.data.hidden == false
        && request.resource.data.joins == 0
        && request.resource.data.likes == 0
        && request.resource.data.comments == 0
        && request.resource.data.completions == 0
        && request.resource.data.metric == 'manual'
        && request.resource.data.target == 1
        && request.resource.data.xpReward == 25
        && request.resource.data.description is string
        && request.resource.data.description.size() >= 10
        && request.resource.data.description.size() <= 500
        && request.resource.data.goal is string
        && request.resource.data.goal.size() >= 5
        && request.resource.data.goal.size() <= 240
        && request.resource.data.instructions is string
        && request.resource.data.instructions.size() > 0
        && request.resource.data.instructions.size() <= 900
        && request.resource.data.creatorName is string
        && request.resource.data.creatorName.size() > 0
        && request.resource.data.creatorName.size() <= 50
        && request.resource.data.createdAtMs is number
        && request.resource.data.createdAtMs <= request.time.toMillis() + 60000
        && request.resource.data.createdAtMs >= request.time.toMillis() - 600000
        && validChallengeInteraction();

      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['featured','hidden','status']);
      allow delete: if isAdmin();

      match /members/{memberUid} {
        allow read: if signedIn();
        allow create, delete: if owner(memberUid)
          && get(/databases/$(database)/documents/communityTasks/$(taskId)).data.kind == 'challenge';
        allow update: if false;
      }
    }
  }}
